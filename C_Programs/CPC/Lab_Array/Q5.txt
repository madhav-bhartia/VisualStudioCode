#include<stdio.h>

int main(){
    int size = 0;
    printf("Enter the size of the array: ");
    scanf("%d", &size);
    int nums[size];
    for(int i = 0; i < size; i++){
        printf("Enter number %d: ", i + 1);
        scanf("%d", &nums[i]);
    }
    int evenCount = 0, oddCount = 0;
    for(int i = 0; i < size; i++){
        if(nums[i] % 2 == 0){
            evenCount++;
        } else {
            oddCount++;
        }
    }
    printf("Even numbers: %d\n", evenCount);
    printf("Odd numbers: %d\n", oddCount);
}
